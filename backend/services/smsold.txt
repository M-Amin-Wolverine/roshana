    // ═══════════════════════════════════════════════════════════════
    // 📤 Send with Kavenegar
    // ═══════════════════════════════════════════════════════════════

    async sendWithKavenegar(phone, message, sender = null) {
        const url = `https://api.kavenegar.com/v1/${CONFIG.apiKey}/sms/send.json`;
        
        const response = await axios.post(url, {
            receptor: phone,
            sender: sender || CONFIG.sender,
            message: message
        }, {
            timeout: 10000,
            headers: {
                'Content-Type': 'application/x-www-form-urlencoded'
            }
        });
        
        if (response.data.return && response.data.return.status !== 200) {
            throw new Error(`Kavenegar error: ${response.data.return.message}`);
        }
        
        return {
            messageId: response.data.entries[0]?.messageid,
            status: response.data.entries[0]?.status,
            statusText: response.data.entries[0]?.statustext
        };
    }

    /**
     * Send bulk SMS
     * @param {Array} recipients - Array of phone numbers
     * @param {string} message - Message content
     * @returns {Promise<Array>} Results
     */
    async sendBulk(recipients, message) {
        const results = [];
        const batchSize = CONFIG.rateLimit.maxPerMinute;
        
        log.info(`Sending bulk SMS to ${recipients.length} recipients`);
        
        // Process in batches
        for (let i = 0; i < recipients.length; i += batchSize) {
            const batch = recipients.slice(i, i + batchSize);
            
            const batchResults = await Promise.allSettled(
                batch.map(phone => this.send({ to: phone, message }))
            );
            
            results.push(...batchResults.map((result, index) => ({
                phone: batch[index],
                ...(result.status === 'fulfilled' ? result.value : { error: result.reason.message })
            })));
            
            // Rate limiting delay between batches
            if (i + batchSize < recipients.length) {
                await this.delay(60000); // Wait 1 minute between batches
            }
        }
        
        return results;
    }

        const scheduledMessage = {
            id: scheduleId,

            status: 'scheduled'

            } catch (error) {

    /**
     * Format phone number to Iranian format
     * @param {string} phone - Phone number
     * @returns {string} Formatted phone
     */
    formatPhoneNumber(phone) {
        if (!phone) return null;
        
        // Remove all non-digit characters
        let cleaned = phone.replace(/\D/g, '');
        
        // Handle different formats
        if (cleaned.startsWith('+98')) {
            cleaned = cleaned.substring(3);
        } else if (cleaned.startsWith('98')) {
            cleaned = cleaned.substring(2);
        } else if (cleaned.startsWith('0')) {
            cleaned = cleaned.substring(1);
        }
        
        // Ensure it starts with 9 (Iranian mobile)
        if (!cleaned.startsWith('9')) {
            cleaned = '9' + cleaned;
        }
        
        return '98' + cleaned;
    }

    /**
     * Check rate limit
     * @param {string} phone - Phone number
     */
    async checkRateLimit(phone) {
        const now = Date.now();
        const key = `sms_${phone}`;
        
        // Get existing data
        let data = this.rateLimitStore.get(key) || {
            minute: { count: 0, resetAt: now + 60000 },
            hour: { count: 0, resetAt: now + 3600000 },
            day: { count: 0, resetAt: now + 86400000 }
        };
        
        // Reset counters if expired
        if (now > data.minute.resetAt) {
            data.minute = { count: 0, resetAt: now + 60000 };
        }
        if (now > data.hour.resetAt) {
            data.hour = { count: 0, resetAt: now + 3600000 };
        }
        if (now > data.day.resetAt) {
            data.day = { count: 0, resetAt: now + 86400000 };
        }
        
        // Check limits
        if (data.minute.count >= CONFIG.rateLimit.maxPerMinute) {
            throw new Error('Rate limit exceeded: max per minute');
        }
        if (data.hour.count >= CONFIG.rateLimit.maxPerHour) {
            throw new Error('Rate limit exceeded: max per hour');
        }
        if (data.day.count >= CONFIG.rateLimit.maxPerDay) {
            throw new Error('Rate limit exceeded: max per day');
        }
        
        // Increment counters
        data.minute.count++;
        data.hour.count++;
        data.day.count++;
        
        this.rateLimitStore.set(key, data);
    }

    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));

    /**
     * Get message length (Unicode aware)
     * @param {string} message - Message
     * @returns {number} Length
     */
    getMessageLength(message) {
        // GSM-7 single segment = 160 chars
        // Unicode (Farsi) = 70 chars per segment
        const isUnicode = /[\u0600-\u06FF\u0750-\u077F\uFB50-\uFDFF\uFE70-\uFEFF]/.test(message);
        return isUnicode ? Math.ceil(message.length / 70) : Math.ceil(message.length / 160);
    }

    /**
     * Split long message into segments
     * @param {string} message - Message
     * @returns {Array<string>} Segments
     */
    splitMessage(message) {
        const isUnicode = /[\u0600-\u06FF\u0750-\u077F\uFB50-\uFDFF\uFE70-\uFEFF]/.test(message);
        const segmentLength = isUnicode ? 67 : 153; // Leave room for concatenation header
        
        const segments = [];
        for (let i = 0; i < message.length; i += segmentLength) {
            segments.push(message.substring(i, i + segmentLength));
        }
        
        return segments;
    }
